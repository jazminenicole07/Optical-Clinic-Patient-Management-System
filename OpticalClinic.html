<script type="text/javascript">
// ARRAYS TO HOLD DATA (LOAD FROM LOCALSTORAGE IF AVAILABLE, ELSE INITIALIZE)
var patients = JSON.parse(localStorage.getItem("patientsData")) || [];
var defaultServices = [
{ name: "Eye Check-up", cost: 500 },
{ name: "Cataract Screening", cost: 800 },
{ name: "Glaucoma Screening", cost: 800 },
{ name: "Dry Eye Assessment", cost: 600 },
{ name: "Ishihara Test", cost: 400 }
];
var services = JSON.parse(localStorage.getItem("servicesData")) || defaultServices;
var consultations = JSON.parse(localStorage.getItem("consultationsData")) || [];
// FUNCTION TO SAVE ARRAYS AS JSON IN LOCALSTORAGE
function saveToLocalStorage() {
localStorage.setItem("patientsData", JSON.stringify(patients));
localStorage.setItem("servicesData", JSON.stringify(services));
localStorage.setItem("consultationsData", JSON.stringify(consultations));
}
// NAVIGATION FUNCTION
function showSection(sectionId) {
document.getElementById("patientSection").style.display = "none";
document.getElementById("serviceSection").style.display = "none";
document.getElementById("consultSection").style.display = "none";
document.getElementById("recordsSection").style.display = "none";
document.getElementById(sectionId).style.display = "block";
// Refresh dropdowns & displays
updateDropdowns();
displayServices();
displayRecords();
}
// 1. ADD / EDIT PATIENT (CREATE / UPDATE)
function savePatient() {
var pId = document.getElementById("pId").value;
if (pId === "") {
alert("Please enter Patient ID!");
return;
}
var pName = document.getElementById("pName").value;
if (pName === "") {
alert("Please enter Full Name!");
return;
}
var pAge = document.getElementById("pAge").value;
if (pAge < 1) {
alert("Please enter a valid age!");
return;
}
var pGender = document.getElementById("pGender").value;
if (pGender === "") {
alert("Please select gender!");
return;
}
var pContact = document.getElementById("pContact").value;
if (pContact === "") {
alert("Please enter contact number!");
return;
}
var pPwd = document.getElementById("pPwd").value;
if (pPwd === "") {
pPwd = "N/A";
}
var pAddress = document.getElementById("pAddress").value;
// Check if patient ID already exists
var existingIndex = -1;
for (var i = 0; i < patients.length; i++) {
if (patients[i].id === pId) {
existingIndex = i;
break;
}
}
var confirmed = confirm("Save record for Patient: " + pName + " (" + pId + ")?");
if (!confirmed) {
alert("Cancelled");
return;
}
if (existingIndex !== -1) {
// Update Record
patients[existingIndex].name = pName;
patients[existingIndex].age = pAge;
patients[existingIndex].gender = pGender;
patients[existingIndex].contact = pContact;
patients[existingIndex].pwd = pPwd;
patients[existingIndex].address = pAddress;
alert("Patient record updated successfully!");
} else {
// Create Record (Push object to array)
var newPatient = {
id: pId,
name: pName,
age: pAge,
gender: pGender,
contact: pContact,
pwd: pPwd,
address: pAddress
};
patients.push(newPatient);
alert("Patient registered successfully!");
}
// Save to LocalStorage
saveToLocalStorage();
// Clear input form
document.getElementById("pId").value = "";
document.getElementById("pName").value = "";
document.getElementById("pAge").value = "";
document.getElementById("pGender").value = "";
document.getElementById("pContact").value = "";
document.getElementById("pPwd").value = "";
document.getElementById("pAddress").value = "";
updateDropdowns();
displayRecords();
}
// 2. ADD & DISPLAY SERVICES
function addService() {
var sName = document.getElementById("sName").value;
var sCost = Number(document.getElementById("sCost").value);
if (sName === "") {
alert("Please enter service name.");
return;
}
if (sCost <= 0) {
alert("Please enter a valid cost.");
return;
}
services.push({ name: sName, cost: sCost });
saveToLocalStorage();
alert("New service added successfully!");
document.getElementById("sName").value = "";
document.getElementById("sCost").value = "";
displayServices();
updateDropdowns();
}
function displayServices() {
var text = "<table><tr><th>Service Name</th><th>Cost (PHP)</th></tr>";
for (var i = 0; i < services.length; i++) {
text += "<tr><td>" + services[i].name + "</td><td>PHP " + Number(services[i].cost).toFixed(2) + "</td></tr>";
}
text += "</table>";
document.getElementById("serviceOutput").innerHTML = text;
}
// 3. CONSULTATION & BILLING PROCESS
function calculateBilling() {
var selectedServiceName = document.getElementById("selectService").value;
var discount = document.getElementById("cDiscount").value;
var paid = Number(document.getElementById("cPaid").value);
var baseCost = 0;
for (var i = 0; i < services.length; i++) {
if (services[i].name === selectedServiceName) {
baseCost = services[i].cost;
break;
}
}
var totalDue = baseCost;
if (discount === "Yes") {
totalDue = baseCost * 0.80; // 20% discount
}
var balance = totalDue - paid;
if (balance < 0) {
balance = 0;
}
document.getElementById("cDue").value = totalDue.toFixed(2);
document.getElementById("cBalance").value = balance.toFixed(2);
}
function processConsultation() {
var pId = document.getElementById("selectPatient").value;
if (pId === "") {
alert("Please select a patient!");
return;
}
var cDate = document.getElementById("cDate").value;
if (cDate === "") {
alert("Please select visit date!");
return;
}
var cReason = document.getElementById("cReason").value;
var cFindings = document.getElementById("cFindings").value;
var service = document.getElementById("selectService").value;
if (service === "") {
alert("Please select service provided!");
return;
}
var due = Number(document.getElementById("cDue").value);
var paid = Number(document.getElementById("cPaid").value);
var balance = Number(document.getElementById("cBalance").value);
var method = document.getElementById("cMethod").value;
var confirmed = confirm("Confirm Billing Process?\nTotal Due: PHP " + due + "\nPaid: PHP " + paid);
if (!confirmed) {
alert("Cancelled");
return;
}
var newConsultation = {
pId: pId,
date: cDate,
reason: cReason,
findings: cFindings,
service: service,
due: due,
paid: paid,
balance: balance,
method: method
};
consultations.push(newConsultation);
saveToLocalStorage();
alert("Consultation and billing record saved!");
document.getElementById("cReason").value = "";
document.getElementById("cFindings").value = "";
document.getElementById("cPaid").value = "";
calculateBilling();
displayRecords();
}
// 4. DROPDOWNS & DATABASE RECORDS (READ, DELETE BY INDEX, SHIFT, POP)
function updateDropdowns() {
var patientSelect = document.getElementById("selectPatient");
patientSelect.innerHTML = "<option value=''>--Select Patient--</option>";
for (var i = 0; i < patients.length; i++) {
patientSelect.innerHTML += "<option value='" + patients[i].id + "'>" + patients[i].id + " - " + patients[i].name + "</option>";
}
var serviceSelect = document.getElementById("selectService");
serviceSelect.innerHTML = "<option value=''>--Select Service--</option>";

}
function displayRecords() {
var search = document.getElementById("searchBar").value.toLowerCase();
// Dashboard summary
var totalRev = 0;
for (var k = 0; k < consultations.length; k++) {
totalRev += Number(consultations[k].paid);
}
document.getElementById("dashboardSummary").innerHTML =
"<b>Total Patients: </b>" + patients.length + "<br>" +
"<b>Total Consultations: </b>" + consultations.length + "<br>" +
"<b>Total Revenue Collected: </b>PHP " + totalRev.toFixed(2);
// Table Display
var html = "<table>" +
"<tr><th>Patient Information</th><th>Consultation & Exam History</th><th>Billing History</th><th>Action</th></tr>";
var count = 0;
for (var i = 0; i < patients.length; i++) {
var p = patients[i];
if (p.id.toLowerCase().indexOf(search) !== -1 || p.name.toLowerCase().indexOf(search) !== -1) {
count++;
// Get patient consults
var consultText = "";
var billingText = "";
for (var j = 0; j < consultations.length; j++) {
if (consultations[j].pId === p.id) {
consultText += "<b>Date:</b> " + consultations[j].date + "<br>" +
"<b>Reason:</b> " + consultations[j].reason + "<br>" +
"<b>Findings:</b> " + consultations[j].findings + "<hr>";
billingText += "<b>Service:</b> " + consultations[j].service + "<br>" +
"<b>Due:</b> PHP " + consultations[j].due + "<br>" +
"<b>Paid:</b> PHP " + consultations[j].paid + " (" + consultations[j].method + ")<br>" +
"<b>Bal:</b> PHP " + consultations[j].balance + "<hr>";
}
}
if (consultText === "") consultText = "<i>No history</i>";
if (billingText === "") billingText = "<i>No billing</i>";
html += "<tr>" +
"<td>" +
"<b>ID:</b> " + p.id + "<br>" +
"<b>Name:</b> " + p.name + "<br>" +
"<b>Age/Gender:</b> " + p.age + " / " + p.gender + "<br>" +
"<b>Contact:</b> " + p.contact + "<br>" +
"<b>PWD/Senior:</b> " + p.pwd + "<br>" +
"<b>Address:</b> " + p.address +
"</td>" +
"<td>" + consultText + "</td>" +
"<td>" + billingText + "</td>" +
"<td>" +
"<input type='button' class='btn btn-edit' value='Edit' onclick='editPatientIndex(" + i + ")'><br><br>" +
"<input type='button' class='btn btn-danger' value='Delete' onclick='deletePatientIndex(" + i + ")'>" +
"</td>" +
"</tr>";
}
}
if (count === 0) {
html += "<tr><td colspan='4'>No patient records found.</td></tr>";
}
html += "</table>";
document.getElementById("output").innerHTML = html;
}

function deleteFirstPatient() {
if (patients.length > 0) {
var deleted = patients.shift();
// Filter consultations to remove entries belonging to the deleted patient
var newConsultations = [];
for (var i = 0; i < consultations.length; i++) {
if (consultations[i].pId !== deleted.id) {
newConsultations.push(consultations[i]);
}
}
consultations = newConsultations;
saveToLocalStorage();
alert("Deleted First Patient and history: " + deleted.name);
displayRecords();
updateDropdowns();
} else {
alert("There is no item in the array to delete.");
}
}
function deleteLastPatient() {

// Filter consultations to remove entries belonging to the deleted patient
var newConsultations = [];
for (var i = 0; i < consultations.length; i++) {

}
}
consultations = newConsultations;
saveToLocalStorage();
alert("Deleted Last Patient and history: " + deleted.name);
displayRecords();
updateDropdowns();
} else {

}
}
function deletePatientIndex(index) {
var deletedPatient = patients[index];
var answer = confirm("Delete patient " + deletedPatient.name + " at Array Index " + index + "?");
if (answer) {
// Remove patient
patients.splice(index, 1);
// Filter consultations to remove entries belonging to the deleted patient
var newConsultations = [];
for (var i = 0; i < consultations.length; i++) {
if (consultations[i].pId !== deletedPatient.id) {
newConsultations.push(consultations[i]);
}
}
consultations = newConsultations;
saveToLocalStorage();
alert("Deleted Item and history: " + deletedPatient.name);
displayRecords();
updateDropdowns();
}
}
